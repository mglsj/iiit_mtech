# ===================== TEST CASES =====================
# <...> means: copy the SHA printed by an earlier step and paste it there.

# 1  init
./mygit init
ls -a

# 2  hash-object
./mygit hash-object test.txt
./mygit hash-object -w test.txt

# 3  cat-file
./mygit cat-file -p 95d09f2b10159347eece71399a7e2e907ea3df4f
./mygit cat-file -s 95d09f2b10159347eece71399a7e2e907ea3df4f
./mygit cat-file -t 95d09f2b10159347eece71399a7e2e907ea3df4f

# 4  write-tree
./mygit write-tree

# 5  cat-file on the tree
./mygit cat-file -t 1ac323ab2a1529daafdd784cba7558ef3575a31d 

# 6  ls-tree
./mygit ls-tree 1ac323ab2a1529daafdd784cba7558ef3575a31d 

# 7  ls-tree --name-only
./mygit ls-tree --name-only 1ac323ab2a1529daafdd784cba7558ef3575a31d

# 8  add (all files)
./mygit add .
cat .mygit/index

# 9  add (specific files)
echo "int main() {}" > main.cpp
echo "int add();" > utils.cpp
./mygit add main.cpp utils.cpp
cat .mygit/index

# 10 commit with message
./mygit commit -m "First commit"

# Change some files
echo "version 2" > test.txt
rm -r test_tree_2
echo "new" > new.txt

# 11 add + commit without message
./mygit add .
./mygit commit

# 12 log
./mygit log

# 13 checkout old commit
./mygit checkout <C1>
ls
cat test.txt

# 14 checkout latest commit
./mygit checkout <C2>
ls
cat test.txt
